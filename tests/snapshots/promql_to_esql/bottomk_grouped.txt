TS metrics-*
| WHERE @timestamp >= ?_tstart AND @timestamp <= ?_tend
| WHERE foo_total IS NOT NULL
| STATS _bucket_value = SUM(RATE(foo_total)) BY time_bucket = TBUCKET(5 minute), service.name
| SORT time_bucket ASC
| STATS value = LAST(_bucket_value, time_bucket) BY service.name
| KEEP service.name, value
| SORT value ASC
| LIMIT 3
