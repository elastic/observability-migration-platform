FROM metrics-*
| WHERE @timestamp >= ?_tstart AND @timestamp <= ?_tend
| WHERE phase == "Running"
| WHERE kube_pod_status_phase IS NOT NULL
| STATS kube_pod_status_phase = AVG(kube_pod_status_phase) BY time_bucket = BUCKET(@timestamp, 50, ?_tstart, ?_tend)
| WHERE kube_pod_status_phase > 0
| SORT time_bucket ASC
