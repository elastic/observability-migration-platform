source: stddev by (job) (http_request_duration_seconds)
feasibility: feasible
---
FROM metrics-*
| WHERE @timestamp >= ?_tstart AND @timestamp <= ?_tend
| WHERE http_request_duration_seconds IS NOT NULL
| STATS http_request_duration_seconds = STD_DEV(http_request_duration_seconds) BY time_bucket = BUCKET(@timestamp, 50, ?_tstart, ?_tend), service.name
| SORT time_bucket ASC
