FROM metrics-*
| WHERE @timestamp >= ?_tstart AND @timestamp <= ?_tend
| WHERE node_memory_MemTotal_bytes IS NOT NULL OR node_memory_MemAvailable_bytes IS NOT NULL
| STATS node_memory_MemTotal_bytes = AVG(node_memory_MemTotal_bytes), node_memory_MemAvailable_bytes = AVG(node_memory_MemAvailable_bytes) BY time_bucket = BUCKET(@timestamp, 50, ?_tstart, ?_tend)
| EVAL computed_value = (node_memory_MemTotal_bytes - node_memory_MemAvailable_bytes)
| KEEP time_bucket, computed_value
| SORT time_bucket ASC
