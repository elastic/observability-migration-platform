FROM metrics-*
| WHERE @timestamp >= ?_tstart AND @timestamp <= ?_tend
| WHERE node_filesystem_avail_bytes IS NOT NULL
| STATS node_filesystem_avail_bytes = PERCENTILE(node_filesystem_avail_bytes, 95) BY time_bucket = BUCKET(@timestamp, 50, ?_tstart, ?_tend), service.name
| SORT time_bucket ASC
