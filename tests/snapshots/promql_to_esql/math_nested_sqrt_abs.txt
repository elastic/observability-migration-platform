| WHERE @timestamp >= ?_tstart AND @timestamp <= ?_tend

| STATS node_memory_usage = AVG(node_memory_usage) BY time_bucket = BUCKET(@timestamp, 50, ?_tstart, ?_tend)
| EVAL node_memory_usage = ABS(node_memory_usage)
